```python
"""CRT-style visual effects: Gaussian glow and scanlines.

Both functions are pure wrappers around OpenCV primitives that return a new
BGR frame. Do not mutate inputs.

Pitfall 4: cv2.GaussianBlur requires an odd kernel; we force it via `k | 1`.
"""
from __future__ import annotations

import cv2
import numpy as np

from openarena_perception.visualization.config import RoomConfig


def apply_glow(frame: np.ndarray, cfg: RoomConfig) -> np.ndarray:
    """CRT phosphor bloom: blur + additive composite."""
    if cfg.glow_radius <= 0 or cfg.glow_strength <= 0:
        return frame
    k = cfg.glow_radius | 1  # force odd — Pitfall 4
    blurred = cv2.GaussianBlur(frame, (k, k), 0)
    return cv2.addWeighted(frame, 1.0, blurred, cfg.glow_strength, 0)


def apply_scanlines(frame: np.ndarray, cfg: RoomConfig) -> np.ndarray:
    """Horizontal CRT scanline darkening."""
    if cfg.scanline_spacing <= 0 or cfg.scanline_strength <= 0:
        return frame
    h, w = frame.shape[:2]
    mask = np.ones((h, 1, 1), dtype=np.float32)
    mask[::cfg.scanline_spacing] = 1.0 - cfg.scanline_strength
    out = frame.astype(np.float32) * mask
    return np.clip(out, 0, 255).astype(np.uint8)
```

```python
from __future__ import annotations

from dataclasses import replace

import numpy as np
import pytest

from openarena_perception.visualization import RoomConfig
from openarena_perception.visualization.effects import apply_glow, apply_scanlines


def _cfg(**overrides) -> RoomConfig:
    return replace(RoomConfig(), **overrides)


def test_glow_increases_nonzero_pixel_count() -> None:
    frame = np.zeros((100, 100, 3), dtype=np.uint8)
    frame[50, 50] = (255, 255, 255)
    before = int(np.count_nonzero(frame))
    out = apply_glow(frame, RoomConfig())
    after = int(np.count_nonzero(out))
    assert after > before  # bloom spread the bright pixel


def test_glow_forces_odd_kernel_with_even_radius() -> None:
    # Pitfall 4: cv2.GaussianBlur with even kernel raises. Must not raise here.
    frame = np.zeros((50, 50, 3), dtype=np.uint8)
    out = apply_glow(frame, _cfg(glow_radius=8, glow_strength=0.3))
    assert out.shape == frame.shape
    assert out.dtype == np.uint8


def test_glow_zero_radius_is_noop() -> None:
    frame = np.full((20, 20, 3), 42, dtype=np.uint8)
    out = apply_glow(frame, _cfg(glow_radius=0))
    assert np.array_equal(out, frame)


def test_scanlines_darken_every_nth_row() -> None:
    frame = np.full((30, 30, 3), 200, dtype=np.uint8)
    out = apply_scanlines(frame, _cfg(scanline_spacing=3, scanline_strength=0.5))
    # row 0, 3, 6, ... must be darker than row 1, 2, 4, 5, ...
    for dark_row in (0, 3, 6, 9):
        for light_row in (1, 2, 4, 5):
            assert out[dark_row, 0, 0] < out[light_row, 0, 0]


def test_scanlines_preserve_uint8_dtype() -> None:
    frame = np.full((30, 30, 3), 200, dtype=np.uint8)
    out = apply_scanlines(frame, _cfg(scanline_spacing=2, scanline_strength=0.3))
```

```python
"""HUD overlay rendering — PRD §6.1 exact elements.

Pitfall 10: all text uses cv2.FONT_HERSHEY_PLAIN (compiled into OpenCV,
not loaded from a file). Never reference TrueType or freetype.
"""
from __future__ import annotations

from typing import Any, Dict, Mapping, Optional

import cv2
import numpy as np

from openarena_perception.visualization.config import RoomConfig

FONT = cv2.FONT_HERSHEY_PLAIN
FONT_SCALE = 0.9
FONT_THICK = 1
LINE_H = 14


def _put(frame: np.ndarray, text: str, org: tuple[int, int], color: tuple[int, int, int]) -> None:
    cv2.putText(frame, text, org, FONT, FONT_SCALE, color, FONT_THICK, cv2.LINE_AA)


def _draw_corner_brackets(frame: np.ndarray, cfg: RoomConfig) -> None:
    h, w = frame.shape[:2]
    L = 20  # bracket arm length
    I = 3   # inset
    c = cfg.grid_color
    # top-left
    cv2.line(frame, (I, I), (I + L, I), c, 1, cv2.LINE_AA)
    cv2.line(frame, (I, I), (I, I + L), c, 1, cv2.LINE_AA)
    # top-right
    cv2.line(frame, (w - I, I), (w - I - L, I), c, 1, cv2.LINE_AA)
    cv2.line(frame, (w - I, I), (w - I, I + L), c, 1, cv2.LINE_AA)
    # bottom-left
    cv2.line(frame, (I, h - I), (I + L, h - I), c, 1, cv2.LINE_AA)
    cv2.line(frame, (I, h - I), (I, h - I - L), c, 1, cv2.LINE_AA)
    # bottom-right
    cv2.line(frame, (w - I, h - I), (w - I - L, h - I), c, 1, cv2.LINE_AA)
    cv2.line(frame, (w - I, h - I), (w - I, h - I - L), c, 1, cv2.LINE_AA)


def _draw_crosshair(frame: np.ndarray, cfg: RoomConfig) -> None:
    cx, cy = cfg.width // 2, cfg.height // 2
    cv2.line(frame, (cx - 5, cy), (cx + 5, cy), cfg.accent_color, 1, cv2.LINE_AA)
    cv2.line(frame, (cx, cy - 5), (cx, cy + 5), cfg.accent_color, 1, cv2.LINE_AA)
    # Guarantee center pixel is accent (AA line may soften it)
    frame[cy, cx] = cfg.accent_color


def render_hud(
    frame: np.ndarray,
    cfg: RoomConfig,
    state: Optional[Mapping[str, Any]] = None,
    extra_hud: Optional[Mapping[str, Any]] = None,
) -> np.ndarray:
    """Draw PRD §6.1 HUD overlay onto frame in place. Returns frame."""
    if not cfg.hud:
        return frame
    state = state or {}
    h, w = frame.shape[:2]

    # ── Top-left fixed block ────────────────────────────────────────
    x0 = 8
    y = 16
    txt_color = cfg.grid_color

    _put(frame, f"FRAME {int(state.get('frame', 0)):06d}", (x0, y), txt_color); y += LINE_H
    _put(
        frame,
        f"ROOM {cfg.room_width:g}x{cfg.room_height:g}x{cfg.room_depth:g}",
        (x0, y),
        txt_color,
    ); y += LINE_H
    _put(
        frame,
        f"FOV {cfg.fov_deg:g} GRID {cfg.grid_lines_x}x{cfg.grid_lines_y}x{cfg.grid_lines_z}",
        (x0, y),
        txt_color,
    ); y += LINE_H
    _put(frame, f"MODE: {state.get('mode', 'DEMO')}", (x0, y), cfg.accent_color); y += LINE_H
    _put(frame, f"OBJECTS: {int(state.get('object_count', 0))}", (x0, y), txt_color); y += LINE_H

    # ── Extra HUD fields (bot state integration point) ──────────────
    if extra_hud:
        y += 4
        for i, (k, v) in enumerate(extra_hud.items()):
            if i >= 10:
                break
            _put(frame, f"{k}: {v}", (x0, y), cfg.accent_color)
            y += LINE_H

    # ── Crosshair ───────────────────────────────────────────────────
    _draw_crosshair(frame, cfg)

    # ── Bottom-left status bar ──────────────────────────────────────
    status = state.get("status_text", "// SPATIAL GRID ACTIVE")
    _put(frame, status, (x0, h - 10), cfg.accent_color)

    # ── Bottom-right source resolution ──────────────────────────────
    src_w = int(state.get("src_width", cfg.width))
    src_h = int(state.get("src_height", cfg.height))
    src_txt = f"SRC {src_w}x{src_h}"
    (tw, th), _ = cv2.getTextSize(src_txt, FONT, FONT_SCALE, FONT_THICK)
    _put(frame, src_txt, (w - tw - 10, h - 10), txt_color)

    # ── Corner brackets ─────────────────────────────────────────────
    _draw_corner_brackets(frame, cfg)

    return frame
```

```python
"""WireframeRoomRenderer — holodeck room frame + scene primitives + HUD.

Takes a SceneGeometry (already in camera space) and produces a BGR frame.
No AAS dependency; no cv2.imshow (Pitfall 7); works with opencv-python-headless.
"""
from __future__ import annotations

from typing import Any, List, Mapping, Optional, Tuple

import cv2
import numpy as np

from openarena_perception.visualization.config import RoomConfig
from openarena_perception.visualization.effects import apply_glow, apply_scanlines
from openarena_perception.visualization.hud import render_hud
from openarena_perception.visualization.projection import (
    cube_edges,
    project_depth_color,
    project_line,
    room_surface_grid,
)
from openarena_perception.visualization.scene import OBJECT_COLORS_BGR, SceneGeometry

Point3 = Tuple[float, float, float]
Segment3 = Tuple[Point3, Point3]


def _build_room_frame_segments(cfg: RoomConfig) -> List[Segment3]:
    """PRD §4.2 — 5 parametric quads tessellated into iso-line segments.

    Room is in CAMERA space: camera at origin looking +Z, +Y down, +X right.
    """
    rw, rh, rd = cfg.room_width, cfg.room_height, cfg.room_depth
    gx, gy, gz = cfg.grid_lines_x, cfg.grid_lines_y, cfg.grid_lines_z

    segs: List[Segment3] = []
    # Floor (at +Y)
    segs += room_surface_grid(
        (-rw / 2, +rh / 2, 0.5), (rw, 0, 0), (0, 0, rd - 0.5), gx, gz
    )
    # Ceiling (at -Y)
    segs += room_surface_grid(
        (-rw / 2, -rh / 2, 0.5), (rw, 0, 0), (0, 0, rd - 0.5), gx, gz
    )
    # Left wall (at -X)
    segs += room_surface_grid(
        (-rw / 2, -rh / 2, 0.5), (0, rh, 0), (0, 0, rd - 0.5), gy, gz
    )
    # Right wall (at +X)
    segs += room_surface_grid(
        (+rw / 2, -rh / 2, 0.5), (0, rh, 0), (0, 0, rd - 0.5), gy, gz
    )
    # Back wall (at +Z)
    segs += room_surface_grid(
        (-rw / 2, -rh / 2, rd), (rw, 0, 0), (0, rh, 0), gx, gy
    )
    return segs


def _seg_midpoint_z(seg: Segment3) -> float:
    return (seg[0][2] + seg[1][2]) / 2.0


class WireframeRoomRenderer:
    """Holodeck-style wireframe renderer.

    Construct once, call `render()` per frame. Room-frame geometry is
    precomputed at init.
    """

    def __init__(self, cfg: RoomConfig) -> None:
        self.cfg = cfg
        self._room_segments: List[Segment3] = _build_room_frame_segments(cfg)
        # Sort back-to-front by midpoint z so far lines are drawn first
        # and near lines overwrite them.
        self._room_segments.sort(key=_seg_midpoint_z, reverse=True)
        self._frame_counter = 0

    def render(
        self,
        bot_pos: Tuple[float, float, float],
        bot_yaw: float,
        scene: SceneGeometry,
        extra_hud: Optional[Mapping[str, Any]] = None,
        state: Optional[Mapping[str, Any]] = None,
    ) -> np.ndarray:
        cfg = self.cfg
        frame = np.full((cfg.height, cfg.width, 3), cfg.bg_color, dtype=np.uint8)

        # 1. Room frame (already camera space)
        self._draw_segments(frame, self._room_segments, fixed_color=None)

        # 2. Scene walls (camera space — sort back-to-front)
        scene_walls_sorted = sorted(scene.walls, key=_seg_midpoint_z, reverse=True)
        self._draw_segments(frame, scene_walls_sorted, fixed_color=None)

        # 3. Scene boxes (cubes — per-class color)
        for box in scene.boxes:
            cls, bx, by, bz, bw, bh, bd = box
            color = OBJECT_COLORS_BGR.get(cls, OBJECT_COLORS_BGR["unknown"])
            edges = cube_edges(bx, by, bz, bw, bh, bd)
            self._draw_segments(frame, list(edges), fixed_color=color)

        # 4. Effects
        if cfg.glow:
            frame = apply_glow(frame, cfg)
        if cfg.scanlines:
            frame = apply_scanlines(frame, cfg)

        # 5. HUD
        if cfg.hud:
            merged_state = dict(state or {})
            merged_state.setdefault("frame", self._frame_counter)
            merged_state.setdefault(
                "object_count", len(scene.walls) + len(scene.boxes)
            )
            merged_state.setdefault("pos", tuple(bot_pos))
            merged_state.setdefault("yaw", bot_yaw)
            render_hud(frame, cfg, merged_state, extra_hud)

        self._frame_counter += 1
        return frame

    def _draw_segments(
        self,
        frame: np.ndarray,
        segments,
        fixed_color: Optional[Tuple[int, int, int]],
    ) -> None:
        cfg = self.cfg
        for seg in segments:
            sp = project_line(seg[0], seg[1], cfg)
            if sp is None:
                continue
            (sx1, sy1), (sx2, sy2) = sp
            if fixed_color is not None:
                color = fixed_color
            else:
                color = project_depth_color(_seg_midpoint_z(seg), cfg)
            cv2.line(
                frame,
                (int(sx1), int(sy1)),
                (int(sx2), int(sy2)),
                color,
                1,
                cv2.LINE_AA,
            )
```
