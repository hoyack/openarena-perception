```python
from enum import StrEnum
from collections import deque
from dataclasses import dataclass, field
from typing import Deque, Optional, Tuple


class MacroIntent(StrEnum):
    """High-level bot intent categories for Phase 10 intent tracking."""
    IDLE = "IDLE"
    TOURING = "TOURING"
    CLIMBING_STAIRS = "CLIMBING_STAIRS"
    AVOIDING_WALL = "AVOIDING_WALL"
    RECOVERING_STUCK = "RECOVERING_STUCK"
    REASSESSING_POSITION = "REASSESSING_POSITION"
    RESPAWNING = "RESPAWNING"
    PAUSING_TO_ASSESS = "PAUSING_TO_ASSESS"


class IntentStatus(StrEnum):
    PLANNING = "planning"
    EXECUTING = "executing"
    COMPLETED = "completed"
    FAILED = "failed"


@dataclass(frozen=True, slots=True)
class Intent:
    """Immutable snapshot of a single bot intent.

    macro: MacroIntent enum value (string-valued)
    sub:   free-form short description ("exiting through door")
    description: longer human-readable description for HUD
    started_at: wall-clock timestamp (time.time()) when this intent began
    target: optional (x,y,z) world position this intent is aimed at
    target_area: optional AAS area id
    status: planning | executing | completed | failed

    Use dataclasses.replace() to create modified copies (intent is frozen).
    """
    macro: MacroIntent
    sub: str
    description: str
    started_at: float
    target: Optional[Tuple[float, float, float]] = None
    target_area: Optional[int] = None
    status: IntentStatus = IntentStatus.EXECUTING

    def duration(self, now: float) -> float:
        return now - self.started_at


@dataclass(frozen=True, slots=True)
class Problem:
    """Immutable record of a bot problem event (stuck, pin, halt, respawn)."""
    kind: str   # "stuck" | "pin" | "wall_halt" | "advisor_halt" | "respawn"
    location: Tuple[float, float, float]
    timestamp: float
    description: str


@dataclass(slots=True)   # NOT frozen — holds mutable deques (Research Pitfall 2)
class BotContext:
    """Mutable container for bot self-awareness state.

    Owned by ItemTour for Phase 10. Designed to be subsystem-neutral so
    StrategicBrain or other loops can adopt it in later phases without
    refactoring the data model.
    """
    current_intent: Intent
    tick_count: int = 0
    path_progress: Tuple[int, int] = (0, 0)
    intent_history: Deque[Intent] = field(default_factory=lambda: deque(maxlen=5))
    problem_history: Deque[Problem] = field(default_factory=lambda: deque(maxlen=5))
    current_area_history: Deque[int] = field(default_factory=lambda: deque(maxlen=10))

    def push_intent(self, new_intent: Intent) -> None:
        """Push current intent to history and replace with new."""
        self.intent_history.append(self.current_intent)
        self.current_intent = new_intent

    def record_problem(self, problem: Problem) -> None:
        self.problem_history.append(problem)

    def record_area(self, area_id: int) -> None:
        if area_id <= 0:
            return
        if self.current_area_history and self.current_area_history[-1] == area_id:
            return  # same area, do not duplicate
        self.current_area_history.append(area_id)
```

```python
"""Pure intent transition detection (Phase 10).

Extracted from ItemTour._on_state so transition logic can be tested without
constructing the full tour object. The rules are edge-triggered (Pitfall 3):
only return a new Intent when the macro or sub actually changes.
"""
from __future__ import annotations

from dataclasses import dataclass
from typing import Optional

from openarena_perception.models import Intent, IntentStatus, MacroIntent


@dataclass
class TransitionSignals:
    """Snapshot of all tick-level signals the transition detector needs.

    This is a plain dataclass, not frozen — callers build and mutate it
    freely each tick before calling detect_intent_transition().
    """
    # Nav state
    nav_is_stuck: bool = False
    nav_is_climbing_stairs: bool = False
    nav_current_area: int = 0
    # Route state
    route_idx: int = 0
    route_target_name: str = ""
    route_target_pos: Optional[tuple] = None
    # Health / respawn
    health: Optional[int] = None
    was_alive_prev_tick: bool = True
    # Halt / wall / pin signals (already computed by item_tour)
    halt_reason: Optional[str] = None
    advisor_hazard: bool = False
    advisor_warning: str = ""
    pinned: bool = False
    pin_escalated: bool = False
    # Time
    now: float = 0.0
    # Previous intent macro+sub for edge-trigger check
    prev_macro: MacroIntent = MacroIntent.IDLE
    prev_sub: str = ""


def detect_intent_transition(
    signals: TransitionSignals,
) -> Optional[Intent]:
    """Return a new Intent if a transition should fire, else None.

    Priority order (highest first):
        1. Death (health <= 0)           → RESPAWNING
        2. Respawn (was_dead → alive)    → TOURING ("recovering from respawn")
        3. Navigator stuck               → RECOVERING_STUCK
        4. Pin escalation                → RECOVERING_STUCK
        5. Proactive halt (wall/visual)  → AVOIDING_WALL
        6. Advisor hazard ahead          → REASSESSING_POSITION
        7. Stair-climb active            → CLIMBING_STAIRS
        8. Stair-climb ended             → TOURING (revert)
        9. Route advance (sub change)    → TOURING (new sub only)

    Returns None if none of the above apply OR the detected (macro, sub)
    matches prev_macro/prev_sub exactly (edge-trigger).
    """
    s = signals

    def _maybe(new_macro: MacroIntent, new_sub: str, description: str,
               target=None, status=IntentStatus.EXECUTING) -> Optional[Intent]:
        if new_macro == s.prev_macro and new_sub == s.prev_sub:
            return None
        return Intent(
            macro=new_macro,
            sub=new_sub,
            description=description,
            started_at=s.now,
            target=target,
            status=status,
        )

    # 1. Death
    if s.health is not None and s.health <= 0:
        return _maybe(MacroIntent.RESPAWNING, "waiting for respawn",
                      "Waiting for respawn")

    # 2. Respawn edge (was dead last tick, now alive)
    if (not s.was_alive_prev_tick) and s.health is not None and s.health > 0:
        return _maybe(MacroIntent.TOURING, "recovering from respawn",
                      "Recovering from respawn")

    # 3. Navigator stuck
    if s.nav_is_stuck:
        return _maybe(MacroIntent.RECOVERING_STUCK, "navigator stuck",
                      "Navigator reports stuck — attempting recovery")

    # 4. Pin escalation (pinned or visual_pin halt reason)
    if s.pin_escalated or (s.halt_reason == "visual_pin" and s.pinned):
        return _maybe(MacroIntent.RECOVERING_STUCK, "pin escalation, replanning",
                      "Pin detected — replanning path")

    # 5. Proactive wall halt
    if s.halt_reason is not None and s.halt_reason != "visual_pin":
        return _maybe(MacroIntent.AVOIDING_WALL, s.halt_reason,
                      f"Avoiding wall: {s.halt_reason}")

    # 6. Advisor hazard
    if s.advisor_hazard:
        return _maybe(MacroIntent.REASSESSING_POSITION,
                      s.advisor_warning or "advisor halt",
                      f"Reassessing after advisor warning")

    # 7. Stair-climb active
    if s.nav_is_climbing_stairs:
        return _maybe(MacroIntent.CLIMBING_STAIRS, "climbing",
                      "Climbing stairs")

    # 8. Stair-climb ended (revert to TOURING — fall through to step 9)
    # 9. Route advance (sub-only change inside TOURING)
    new_sub = f"heading to {s.route_target_name}" if s.route_target_name else "touring"
    return _maybe(MacroIntent.TOURING, new_sub,
                  f"Heading to {s.route_target_name}" if s.route_target_name else "Touring",
                  target=s.route_target_pos)
```

```python
# This test module does NOT construct a full ItemTour (too heavy). Instead it
# tests the pause gate logic in isolation by building a minimal harness that
# exercises the same code path. If the ItemTour pause logic becomes complex
# enough to warrant an integration test, migrate to constructing ItemTour with
# mocked bridge/nav/agent.


class PauseGateHarness:
    """Minimal stand-in that mirrors ItemTour's pause gate behavior."""
    def __init__(self, pause_duration: float):
        self._intent_pause_until = 0.0
        self._intent_pause_duration = pause_duration
        self._threat_flag = False
        self.release_all_calls = 0
        # Fake nav + gs
        self.nav_is_stuck = False
        self.health = 100
        # Current intent for status tracking
        self.current_intent = Intent(
            macro=MacroIntent.IDLE, sub="", description="",
            started_at=0.0, status=IntentStatus.EXECUTING,
        )
        # Track whether the gate returned (i.e., tick was paused)
        self.last_paused = False

    @property
    def threat_flag(self) -> bool:
        return self._threat_flag

    def release_all(self):
        self.release_all_calls += 1

    def open_pause(self, now: float):
        """Equivalent to a macro transition firing."""
        if not self.threat_flag:
            self._intent_pause_until = now + self._intent_pause_duration
            from dataclasses import replace
            self.current_intent = replace(self.current_intent, status=IntentStatus.PLANNING)
            self.release_all()

    def tick(self, now: float) -> bool:
        """Returns True if the tick was paused (gate returned early)."""
        if now < self._intent_pause_until:
            health_ok = self.health is None or self.health > 0
            if not self.threat_flag and not self.nav_is_stuck and health_ok:
                self.last_paused = True
                return True
            # Emergency break-out
            self._intent_pause_until = 0.0
            if self.current_intent.status == IntentStatus.PLANNING:
                from dataclasses import replace
                self.current_intent = replace(self.current_intent, status=IntentStatus.EXECUTING)
        self.last_paused = False
        return False
```

```python
        description="", started_at=995.0,
    ))
    ctx.problem_history.append(Problem(
        kind="wall_halt", location=(481.0, 1336.0, -93.0),
        timestamp=994.0, description="vision score=0.89",
    ))
    ctx.current_area_history.extend([299, 259, 285, 304, 325])
    return ctx
```
