STEP 5 — Add _draw_intent_panel method

Add a new method to the HUD class. Call signature:

```python
def _draw_intent_panel(self, x_offset: int = INTENT_PANEL_X) -> None:
    """Render the Phase 10 right-side intent panel.

    Layout (top-to-bottom):
      - Header label "INTENT" (bold)
      - Current intent box (with optional pause flash border)
          * Macro name in big bright yellow (font_intent)
          * Sub-intent in white (font_sub)
          * Duration counter "X.Xs"
      - Section label "HISTORY"
      - Intent history (5 rows, dim gray, font_history)
      - Section label "PROBLEMS"
      - Problem history (5 rows, color-coded, font_history)
      - Section label "PATH"
      - Path progress bar with "N/M" text
      - Section label "STATS"
      - Advisor stats + halt counters (from existing HUDState fields)
    """
    s = self._state   # current HUDState
    pad = INTENT_PANEL_PAD
    x = x_offset + pad
    y = 20

    # --- Header ---
    self._text("INTENT", x, y, self.font_label, INTENT_LABEL)
    y += 20

    # --- Current intent box ---
    box_x = x
    box_y = y
    box_w = (HUD_WIDTH - x_offset) - 2 * pad
    box_h = 90
    pygame.draw.rect(self.screen, (40, 40, 50),
                     (box_x, box_y, box_w, box_h), border_radius=4)

    # Flashing blue border during pause-to-assess (2Hz)
    if s.is_paused_for_assess:
        flash_on = int(time.time() * 2) % 2 == 0
        if flash_on:
            pygame.draw.rect(self.screen, PAUSE_BORDER,
                             (box_x, box_y, box_w, box_h),
                             width=3, border_radius=4)

    # Current intent text
    macro = s.intent.get("macro", "IDLE") if s.intent else "IDLE"
    sub = s.intent.get("sub", "") if s.intent else ""
    duration_s = s.intent.get("duration_s", 0.0) if s.intent else 0.0
    self._text(str(macro), box_x + 10, box_y + 8, self.font_intent, INTENT_CURRENT)
    if sub:
        self._text(str(sub), box_x + 10, box_y + 36, self.font_sub, INTENT_SUB)
    self._text(f"{duration_s:.1f}s", box_x + 10, box_y + 60,
               self.font_sub, INTENT_LABEL)

    y = box_y + box_h + 15

    # --- Intent history ---
    self._text("HISTORY", x, y, self.font_label, INTENT_LABEL)
    y += 16
    for item in list(s.intent_history)[-5:]:
        macro = item.get("macro", "?")
        sub = item.get("sub", "")
        dur = item.get("duration_s", 0.0)
        line = f"{macro:<14s} {sub[:18]:<18s} {dur:>4.1f}s"
        self._text(line, x, y, self.font_history, INTENT_HISTORY)
        y += 14

    # Pad to 5 rows visually even when fewer are available
    empty_rows = 5 - min(5, len(s.intent_history))
    y += empty_rows * 14

    y += 10

    # --- Problem history ---
    self._text("PROBLEMS", x, y, self.font_label, INTENT_LABEL)
    y += 16
    problem_colors = {
        "wall_halt": PROBLEM_WALL,
        "stuck": PROBLEM_STUCK,
        "pin": PROBLEM_PIN,
        "respawn": PROBLEM_RESPAWN,
        "advisor_halt": PROBLEM_ADVISOR,
    }
    for prob in list(s.problem_history)[-5:]:
        kind = prob.get("kind", "?")
        desc = prob.get("description", "")
        color = problem_colors.get(kind, (200, 200, 200))
        line = f"{kind:<12s} {desc[:24]}"
        self._text(line, x, y, self.font_history, color)
        y += 14

    empty_rows = 5 - min(5, len(s.problem_history))
    y += empty_rows * 14

    y += 10

    # --- Path progress ---
    self._text("PATH", x, y, self.font_label, INTENT_LABEL)
    y += 16
    cur_step = s.path_progress.get("current_step", 0)
    total_steps = s.path_progress.get("total_steps", 0)
    self._text(f"{cur_step}/{total_steps}", x, y,
               self.font_history, INTENT_SUB)

    # Progress bar (simple rectangle with filled portion)
    bar_y = y + 16
    bar_w = box_w
    bar_h = 8
    pygame.draw.rect(self.screen, (60, 60, 70),
                     (x, bar_y, bar_w, bar_h), border_radius=2)
    if total_steps > 0:
        fill_w = int(bar_w * (cur_step / total_steps))
        pygame.draw.rect(self.screen, INTENT_CURRENT,
                         (x, bar_y, fill_w, bar_h), border_radius=2)
    y = bar_y + bar_h + 15

    # --- Stats (reuse existing HUDState fields for advisor / halts) ---
    self._text("STATS", x, y, self.font_label, INTENT_LABEL)
    y += 16
    # Pull from existing HUDState fields (these were already present before Plan 10)
    advisor_sent = getattr(s, "advisor_sent", None)
    advisor_complete = getattr(s, "advisor_complete", None)
    proactive_halts = getattr(s, "proactive_halts", None)
    if advisor_sent is not None:
        self._text(f"advisor: {advisor_sent}/{advisor_complete}",
                   x, y, self.font_history, INTENT_SUB)
        y += 14
    if proactive_halts is not None:
        self._text(f"halts: {proactive_halts}", x, y,
                   self.font_history, INTENT_SUB)
        y += 14
    # threat flag badge
    threat_color = PROBLEM_WALL if s.threat_flag else (100, 200, 100)
    threat_text = "THREAT" if s.threat_flag else "no threat"
    self._text(threat_text, x, y, self.font_history, threat_color)
```

```python
import json
import sys
from pathlib import Path


def test_contract_build_payload_to_strategic_hud_load(tmp_path):
    """End-to-end contract: build_hud_payload output must be parseable by
    strategic_hud.HUDState.load into a HUDState with populated intent fields.

    This test imports strategic_hud from the repo root (the script is not
    part of the openarena_perception package — it lives at the project root).
    """
    # Ensure the repo root is on sys.path so we can import strategic_hud
    repo_root = Path(__file__).resolve().parents[1]
    if str(repo_root) not in sys.path:
        sys.path.insert(0, str(repo_root))

    import strategic_hud   # noqa: E402

    ctx = _ctx()   # helper defined earlier in this file by Plan 10-02
    payload = build_hud_payload(
        context=ctx,
        map_name="q3dm1",
        is_paused_for_assess=True,
        threat_flag=False,
    )

    # Write to a tmp file (this is what item_tour.py does in prod)
    state_file = tmp_path / "hud_state.json"
    state_file.write_text(json.dumps(payload))

    # Load via the HUD's actual loader
    s = strategic_hud.HUDState.load(state_file)

    # Contract assertions
    assert s.intent, "intent dict should be populated"
    assert s.intent.get("macro") == "TOURING"
    assert s.intent.get("sub") == "heading to item_health_mega"

    assert isinstance(s.intent_history, list)
    assert len(s.intent_history) == 1
    assert s.intent_history[0].get("macro") == "AVOIDING_WALL"

    assert isinstance(s.problem_history, list)
    assert s.problem_history[0].get("kind") == "wall_halt"

    assert s.path_progress.get("current_step") == 5
    assert s.path_progress.get("total_steps") == 13

    assert s.is_paused_for_assess is True
    assert s.threat_flag is False

    assert isinstance(s.area_history, list)
    assert s.area_history == [299, 259, 285, 304, 325]
```
