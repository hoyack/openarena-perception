Run `uv init --no-readme --no-pin-python` in the project root to create pyproject.toml. Then edit pyproject.toml to set:

```toml
[project]
name = "openarena-perception"
version = "0.1.0"
description = "Brain-inspired AI bot framework for OpenArena"
requires-python = ">=3.11"
dependencies = [
    "mss>=10.0",
    "opencv-python-headless>=4.8",
    "pytesseract>=0.3",
    "numpy>=1.26",
    "ollama>=0.6",
    "pynput>=1.7",
    "pygame>=2.5",
    "pydantic>=2.6",
    "pydantic-settings>=2.13",
    "structlog>=24.0",
]

[project.optional-dependencies]
dev = [
    "pytest>=8.0",
    "pytest-cov>=5.0",
    "ruff>=0.15",
]

[tool.ruff]
target-version = "py311"
line-length = 100

[tool.ruff.lint]
select = ["E", "F", "W", "I", "N", "UP", "B"]
ignore = ["E501"]

[tool.ruff.format]
quote-style = "double"

[tool.pytest.ini_options]
testpaths = ["tests"]
python_files = ["test_*.py"]
python_classes = ["Test*"]
python_functions = ["test_*"]
addopts = "-v --tb=short"
```

Then run:
1. `uv add mss opencv-python-headless pytesseract "numpy>=1.26" "ollama>=0.6" "pynput>=1.7" "pygame>=2.5" "pydantic>=2.6" "pydantic-settings>=2.13" "structlog>=24.0"` to add runtime deps
2. `uv add --dev "pytest>=8.0" "pytest-cov>=5.0" "ruff>=0.15"` to add dev deps
3. Verify `uv.lock` was created

Note: Keep requirements.txt for now (do not delete) as a reference. The uv.lock is the new source of truth.
