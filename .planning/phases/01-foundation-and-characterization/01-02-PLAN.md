from __future__ import annotations

from functools import lru_cache
from pathlib import Path
from typing import ClassVar

from pydantic import Field
from pydantic_settings import BaseSettings, SettingsConfigDict, TomlConfigSettingsSource


class BotSettings(BaseSettings):
    """Central configuration — env vars override .env override TOML."""

    model_config = SettingsConfigDict(
        env_prefix="OA_",
        env_file=".env",
        env_file_encoding="utf-8",
        toml_file="bot.toml",
        extra="ignore",
    )

    # Display
    screen_w: int = 800
    screen_h: int = 600

    # Game connection
    pipe_path: str = "/tmp/openarena_pipe"
    rcon_password: str = ""
    rcon_port: int = 27960

    # Perception
    enemy_color_mode: str = "red"
    ocr_upscale: int = 3
    enemy_min_area: int = 200

    # LLM
    ollama_model: str = "gemma3:4b"
    ollama_base_url: str = "http://localhost:11434"
    llm_enabled: bool = True

    # Navigation
    map_name: str = "dm4ish"
    aas_dir: str = "maps"

    # Performance
    tick_budget_ms: int = 50
    target_fps: int = 15

    # Logging
    log_json: bool = True
    log_level: str = "INFO"

    @classmethod
    def settings_customise_sources(cls, settings_cls, init_settings,
                                    env_settings, dotenv_settings,
                                    file_secret_settings):
        return (
            init_settings,
            env_settings,
            dotenv_settings,
            TomlConfigSettingsSource(settings_cls),
        )


@lru_cache
def get_settings() -> BotSettings:
    """Singleton settings instance. Import this everywhere."""
    return BotSettings()

**Step 3: Create `openarena_perception/logging_config.py`** (named logging_config.py to avoid shadowing stdlib logging):

```python
"""Structured logging via structlog with JSON output and subsystem context binding."""
from __future__ import annotations

import logging

import structlog


def configure_logging(json_output: bool = True, level: str = "INFO") -> None:
    """Configure structlog for the entire application. Call once at startup."""
    shared_processors = [
        structlog.contextvars.merge_contextvars,
        structlog.processors.add_log_level,
        structlog.processors.TimeStamper(fmt="iso"),
        structlog.processors.StackInfoRenderer(),
        structlog.processors.format_exc_info,
    ]

    if json_output:
        renderer = structlog.processors.JSONRenderer()
    else:
        renderer = structlog.dev.ConsoleRenderer()

    structlog.configure(
        processors=[
            *shared_processors,
            structlog.stdlib.ProcessorFormatter.wrap_for_formatter,
        ],
        logger_factory=structlog.stdlib.LoggerFactory(),
        wrapper_class=structlog.stdlib.BoundLogger,
        cache_logger_on_first_use=True,
    )

    formatter = structlog.stdlib.ProcessorFormatter(
        processors=[
            structlog.stdlib.ProcessorFormatter.remove_processors_meta,
            renderer,
        ],
    )
    handler = logging.StreamHandler()
    handler.setFormatter(formatter)
    root = logging.getLogger()
    root.handlers.clear()
    root.addHandler(handler)
    root.setLevel(getattr(logging, level.upper()))


def get_logger(subsystem: str) -> structlog.stdlib.BoundLogger:
    """Get a logger pre-bound with subsystem context."""
    return structlog.get_logger(subsystem=subsystem)
```
