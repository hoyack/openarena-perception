```python
rooms: List[Room] = []
# Spatial dedup: corner_key -> (cycle, polygon, selected_corners, perimeter)
best_by_corner_key: Dict[Tuple[int, ...], Tuple[List[int], List[Tuple[float, float]], List[Corner], float]] = {}

for cycle in cycles:
    polygon = self._cycle_to_polygon(cycle, corners)
    if len(polygon) < 3:
        continue
    # Select the Corner objects participating in this cycle (same rule as before).
    selected_corners = [
        c for c in corners
        if c.wall_a_id in cycle and c.wall_b_id in cycle
    ]
    if not selected_corners:
        continue
    corner_key = tuple(sorted(c.id for c in selected_corners))
    perimeter = self._polygon_perimeter(polygon)
    existing = best_by_corner_key.get(corner_key)
    if existing is None or perimeter < existing[3]:
        best_by_corner_key[corner_key] = (cycle, polygon, selected_corners, perimeter)

# Emit rooms from the dedup dict, applying area filter AFTER shortest-perimeter selection.
next_room_id = 1
for corner_key, (cycle, polygon, selected_corners, _perim) in best_by_corner_key.items():
    area = self._polygon_area(polygon)
    if area < self.min_room_area:
        continue
    xs = [p[0] for p in polygon]
    ys = [p[1] for p in polygon]
    bounds = (min(xs), min(ys), max(xs), max(ys))
    rooms.append(
        Room(
            id=next_room_id,
            polygon_vertices=tuple((float(x), float(y)) for x, y in polygon),
            bounds=tuple(float(v) for v in bounds),
            area=float(area),
            wall_ids=tuple(sorted(cycle)),
            corner_ids=tuple(c.id for c in selected_corners),
            doorway_ids=tuple(),
            items_seen=tuple(),
            visit_count=0,
            first_seen_ts=0.0,
            last_seen_ts=0.0,
            last_visited_ts=0.0,
            notes="",
        )
    )
    next_room_id += 1
```

3. Add a `_polygon_perimeter` static helper next to `_polygon_area`:

```python
@staticmethod
def _polygon_perimeter(polygon: List[Tuple[float, float]]) -> float:
    n = len(polygon)
    if n < 2:
        return 0.0
    total = 0.0
    for i in range(n):
        x1, y1 = polygon[i]
        x2, y2 = polygon[(i + 1) % n]
        total += ((x2 - x1) ** 2 + (y2 - y1) ** 2) ** 0.5
    return total
```

```python
def test_room_spatial_dedup():
    """Plan 13-08: two enumerations over the same corner set emit exactly 1 Room."""
    from openarena_perception.navigation.room_perception import (
        CornerDetector,
        RoomInferrer,
        Wall,
    )

    # 200x200 square — 4 corners, one room. With the old wall-id-tuple dedup
    # a diagonal "chord" wall introduced new cycles that shared the corner
    # set. We construct two overlapping walls along one edge to force the
    # cycle enumerator to find >1 cycle through the same 4 corners.
    w1 = Wall(1, 0, 0, 200, 0, 0.9, 30, 0.0, 1.0)
    w2 = Wall(2, 200, 0, 200, 200, 0.9, 30, 0.0, 1.0)
    w3 = Wall(3, 200, 200, 0, 200, 0.9, 30, 0.0, 1.0)
    w4 = Wall(4, 0, 200, 0, 0, 0.9, 30, 0.0, 1.0)
    # Near-duplicate of w1 — touches the same two corners so it creates a
    # second cycle {5,2,3,4} over the same 4-corner set.
    w5 = Wall(5, 0, 1, 200, 1, 0.9, 30, 0.0, 1.0)
    walls = [w1, w2, w3, w4, w5]
    corners = CornerDetector().detect(walls)
    rooms, _ = RoomInferrer().infer(walls, corners)
    # Even though two cycles can be enumerated, the corner set is identical,
    # so spatial dedup collapses them to a single Room.
    assert len(rooms) == 1, f"expected 1 room after spatial dedup, got {len(rooms)}"


def test_room_shortest_perimeter_wins():
```
