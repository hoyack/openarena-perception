    ```python
    """Phase 13: shared geometry helpers for room perception detectors.

    Private module — no public exports. Both corner_detector.py and
    doorway_detector.py import from here so the math lives in exactly one place.
    """
    from __future__ import annotations

    import math
    from typing import Tuple

    from openarena_perception.navigation.room_perception.types import Wall


    def nearest_endpoints(
        w1: Wall, w2: Wall
    ) -> Tuple[Tuple[float, float], Tuple[float, float], float]:
        """Return the closest endpoint pair (e1, e2, distance) between two walls."""
        ends_1 = [(w1.x1, w1.y1), (w1.x2, w1.y2)]
        ends_2 = [(w2.x1, w2.y1), (w2.x2, w2.y2)]
        best = (ends_1[0], ends_2[0], float("inf"))
        for e1 in ends_1:
            for e2 in ends_2:
                d = math.hypot(e1[0] - e2[0], e1[1] - e2[1])
                if d < best[2]:
                    best = (e1, e2, d)
        return best


    def angle_between(w1: Wall, w2: Wall) -> float:
        """Return the unsigned angle (degrees, 0..180) between two wall directions."""
        a1 = math.degrees(math.atan2(w1.y2 - w1.y1, w1.x2 - w1.x1))
        a2 = math.degrees(math.atan2(w2.y2 - w2.y1, w2.x2 - w2.x1))
        return abs(((a1 - a2 + 180) % 360) - 180)
    ```

    ```python
    """Phase 13: Corner detector — pairwise wall intersection gating."""
    from __future__ import annotations

    import itertools
    from typing import List

    from openarena_perception.navigation.room_perception._geometry import (
        angle_between,
        nearest_endpoints,
    )
    from openarena_perception.navigation.room_perception.types import Corner, Wall


    class CornerDetector:
        def __init__(
            self,
            *,
            angle_min: float = 20.0,
            angle_max: float = 160.0,
            endpoint_max_dist: float = 32.0,
        ) -> None:
            self.angle_min = angle_min
            self.angle_max = angle_max
            self.endpoint_max_dist = endpoint_max_dist

        def detect(self, walls: List[Wall]) -> List[Corner]:
            corners: List[Corner] = []
            next_id = 1
            for w1, w2 in itertools.combinations(walls, 2):
                angle = angle_between(w1, w2)
                if not (self.angle_min < angle < self.angle_max):
                    continue
                p1, p2, dist = nearest_endpoints(w1, w2)
                if dist > self.endpoint_max_dist:
                    continue
                mid_x = (p1[0] + p2[0]) / 2
                mid_y = (p1[1] + p2[1]) / 2
                corners.append(
                    Corner(
                        id=next_id,
                        x=float(mid_x),
                        y=float(mid_y),
                        wall_a_id=w1.id,
                        wall_b_id=w2.id,
                        angle_deg=float(angle),
                        confidence=min(w1.confidence, w2.confidence),
                    )
                )
                next_id += 1
            return corners
    ```

    ```python
    """Phase 13: Doorway detector — near-miss wall gaps."""
    from __future__ import annotations

    import itertools
    from typing import List

    from openarena_perception.navigation.room_perception._geometry import (
        angle_between,
        nearest_endpoints,
    )
    from openarena_perception.navigation.room_perception.types import Corner, Doorway, Wall


    class DoorwayDetector:
        def __init__(self, *, gap_min: float = 4.0, gap_max: float = 96.0) -> None:
            self.gap_min = gap_min
            self.gap_max = gap_max

        def detect(
            self, walls: List[Wall], corners: List[Corner] | None = None
        ) -> List[Doorway]:
            # Build set of wall pairs already bridged by a corner — skip them.
            bridged: set = set()
            if corners:
                for c in corners:
                    key = tuple(sorted((c.wall_a_id, c.wall_b_id)))
                    bridged.add(key)

            doorways: List[Doorway] = []
            next_id = 1
            for w1, w2 in itertools.combinations(walls, 2):
                key = tuple(sorted((w1.id, w2.id)))
                if key in bridged:
                    continue
                p1, p2, dist = nearest_endpoints(w1, w2)
                if not (self.gap_min < dist < self.gap_max):
                    continue
                # Only collinear-ish walls count as doorway candidates
                angle = angle_between(w1, w2)
                if 20.0 < angle < 160.0:
                    # Perpendicular-ish, not a doorway — skip
                    continue
                doorways.append(
                    Doorway(
                        id=next_id,
                        wall_a_id=w1.id,
                        wall_b_id=w2.id,
                        gap_distance=float(dist),
                        midpoint_x=float((p1[0] + p2[0]) / 2),
                        midpoint_y=float((p1[1] + p2[1]) / 2),
                    )
                )
                next_id += 1
            return doorways
    ```

    ```python
    """Phase 13: Room inferrer — hand-rolled simple cycle enumeration over corner graph."""
    from __future__ import annotations

    from typing import Dict, List, Set, Tuple

    from openarena_perception.navigation.room_perception.types import (
        Corner,
        PartialRoom,
        Room,
        Wall,
    )


    class RoomInferrer:
        def __init__(
            self, *, min_room_area: float = 4096.0, max_cycle_length: int = 8
        ) -> None:
            self.min_room_area = min_room_area
            self.max_cycle_length = max_cycle_length

        def infer(
            self, walls: List[Wall], corners: List[Corner]
        ) -> Tuple[List[Room], List[PartialRoom]]:
            if not walls:
                return [], []

            # Wall id -> Wall
            wall_by_id: Dict[int, Wall] = {w.id: w for w in walls}

            # Build adjacency: wall -> walls it shares a corner with
            adj: Dict[int, Set[int]] = {w.id: set() for w in walls}
            for c in corners:
                adj[c.wall_a_id].add(c.wall_b_id)
                adj[c.wall_b_id].add(c.wall_a_id)

            # Find simple cycles via DFS, dedup by canonical form
            cycles = self._find_cycles(adj, self.max_cycle_length)

            rooms: List[Room] = []
            used_wall_sets: Set[Tuple[int, ...]] = set()
            next_room_id = 1

            for cycle in cycles:
                key = tuple(sorted(cycle))
                if key in used_wall_sets:
                    continue
                polygon = self._cycle_to_polygon(cycle, corners)
                if len(polygon) < 3:
                    continue
                area = self._polygon_area(polygon)
                if area < self.min_room_area:
                    continue
                used_wall_sets.add(key)
                xs = [p[0] for p in polygon]
                ys = [p[1] for p in polygon]
                bounds = (min(xs), min(ys), max(xs), max(ys))
                rooms.append(
                    Room(
                        id=next_room_id,
                        polygon_vertices=tuple((float(x), float(y)) for x, y in polygon),
                        bounds=tuple(float(v) for v in bounds),
                        area=float(area),
                        wall_ids=tuple(cycle),
                        corner_ids=tuple(
                            c.id for c in corners
                            if c.wall_a_id in cycle and c.wall_b_id in cycle
                        ),
                        doorway_ids=tuple(),
                        items_seen=tuple(),
                        visit_count=0,
                        first_seen_ts=0.0,
                        last_seen_ts=0.0,
                        last_visited_ts=0.0,
                        notes="",
                    )
                )
                next_room_id += 1

            # PartialRoom = any wall chain of length 2-3 that is NOT in a closed room
            partial_rooms: List[PartialRoom] = []
            closed_wall_ids: Set[int] = set()
            for r in rooms:
                closed_wall_ids.update(r.wall_ids)
            next_partial_id = 1
            # Build connected components from adj, restricted to walls not in a room
            visited: Set[int] = set()
            for w_id in adj:
                if w_id in visited or w_id in closed_wall_ids:
                    continue
                component = self._bfs_component(w_id, adj)
                visited.update(component)
                if 2 <= len(component) <= 3:
                    corner_ids = [
                        c.id
                        for c in corners
                        if c.wall_a_id in component and c.wall_b_id in component
                    ]
                    partial_rooms.append(
                        PartialRoom(
                            id=next_partial_id,
                            wall_ids=tuple(sorted(component)),
                            corner_ids=tuple(corner_ids),
                        )
                    )
                    next_partial_id += 1

            # If a room has walls/corners but below min_area, the walls are
            # technically closed — still report them as partial (spec: below-min
            # area = not a renderable room). Handled above by falling through.
            if not rooms and len(walls) >= 2 and not partial_rooms:
                # Fallback: small-area closed shape or any wall chain becomes partial
                all_wall_ids = sorted(w.id for w in walls)
                if 2 <= len(all_wall_ids) <= 8:
                    partial_rooms.append(
                        PartialRoom(
                            id=next_partial_id,
                            wall_ids=tuple(all_wall_ids),
                            corner_ids=tuple(c.id for c in corners),
                        )
                    )

            return rooms, partial_rooms

        # ------------------------------------------------------------------
        def _find_cycles(
            self, adj: Dict[int, Set[int]], max_len: int
        ) -> List[List[int]]:
            """Enumerate simple cycles up to max_len walls. Canonicalize by min-id start."""
            cycles: List[List[int]] = []
            seen: Set[Tuple[int, ...]] = set()
            nodes = sorted(adj.keys())
            for start in nodes:
                stack: List[Tuple[int, List[int]]] = [(start, [start])]
                while stack:
                    node, path = stack.pop()
                    if len(path) > max_len:
                        continue
                    for neighbor in adj.get(node, ()):
                        if neighbor == start and len(path) >= 3:
                            key = tuple(sorted(path))
                            if key not in seen:
                                seen.add(key)
                                cycles.append(list(path))
                        elif neighbor not in path and neighbor > start:
                            stack.append((neighbor, path + [neighbor]))
            return cycles

        def _cycle_to_polygon(
            self, cycle: List[int], corners: List[Corner]
        ) -> List[Tuple[float, float]]:
            """Each consecutive wall pair in the cycle should share a corner;
            the corner position is a polygon vertex."""
            polygon: List[Tuple[float, float]] = []
            n = len(cycle)
            for i in range(n):
                a = cycle[i]
                b = cycle[(i + 1) % n]
                shared = [
                    c for c in corners
                    if {c.wall_a_id, c.wall_b_id} == {a, b}
                ]
                if not shared:
                    return []  # cycle is not corner-connected
                polygon.append((shared[0].x, shared[0].y))
            return polygon

        @staticmethod
        def _polygon_area(polygon: List[Tuple[float, float]]) -> float:
            """Shoelace formula."""
            n = len(polygon)
            if n < 3:
                return 0.0
            s = 0.0
            for i in range(n):
                x1, y1 = polygon[i]
                x2, y2 = polygon[(i + 1) % n]
                s += x1 * y2 - x2 * y1
            return abs(s) / 2.0

        @staticmethod
        def _bfs_component(start: int, adj: Dict[int, Set[int]]) -> List[int]:
            visited: Set[int] = set()
            queue = [start]
            while queue:
                node = queue.pop(0)
                if node in visited:
                    continue
                visited.add(node)
                for n in adj.get(node, ()):
                    if n not in visited:
                        queue.append(n)
            return sorted(visited)
    ```
