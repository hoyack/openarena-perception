    1. Append to `openarena_perception/navigation/aas_parser.py` a new module-level function:
       ```python
       def areas_on_floor(
           graph: "AASGraph",
           bot_z: float,
           z_tolerance: float = 24.0,
       ) -> List[int]:
           """Return area indices whose z-range contains bot_z (± tolerance).

           Skips area 0 (dummy). z_tolerance absorbs quantization noise so the
           bot's current area survives narrow z-bands. Default 24.0 ≈ half a
           Q3 player hull height.
           """
           out: List[int] = []
           lo = bot_z - z_tolerance
           hi = bot_z + z_tolerance
           for i, a in enumerate(graph.areas):
               if i == 0:
                   continue
               amin = a.mins[2]
               amax = a.maxs[2]
               if amax < lo:
                   continue
               if amin > hi:
                   continue
               out.append(i)
           return out
       ```
       Export it by name — no __all__ manipulation needed (module already uses implicit public API).

    2. Create `tests/test_aas_parser.py` (if it does not exist) or append to it. Use the synthetic-AASGraph pattern from `tests/test_property_based.py` — construct `AASArea` objects directly with explicit mins/maxs tuples, wrap in `AASGraph(areas=[...], num_areas=N)`.

       ```python
       from openarena_perception.navigation.aas_parser import (
           AASArea, AASGraph, areas_on_floor, parse_aas
       )

       def _mk_area(idx, z_lo, z_hi):
           return AASArea(
               idx=idx, cx=0.0, cy=0.0, cz=(z_lo + z_hi) / 2,
               mins=(0.0, 0.0, float(z_lo)),
               maxs=(0.0, 0.0, float(z_hi)),
           )

       def _mk_graph(*areas):
           full = [_mk_area(0, 0, 0)] + list(areas)  # dummy area 0
           return AASGraph(areas=full, num_areas=len(full))

       def test_areas_on_floor_includes_area_containing_bot_z():
           g = _mk_graph(_mk_area(1, -10, 10))
           assert areas_on_floor(g, 0.0, z_tolerance=24.0) == [1]

       def test_areas_on_floor_excludes_area_far_above():
           g = _mk_graph(_mk_area(1, 100, 200))
           assert areas_on_floor(g, 0.0, z_tolerance=24.0) == []

       def test_areas_on_floor_excludes_area_far_below():
           g = _mk_graph(_mk_area(1, -200, -100))
           assert areas_on_floor(g, 0.0, z_tolerance=24.0) == []

       def test_areas_on_floor_tolerance_band_inclusive():
           g = _mk_graph(_mk_area(1, 20, 50))  # lower edge 20 <= 0+24
           assert areas_on_floor(g, 0.0, z_tolerance=24.0) == [1]

       def test_areas_on_floor_skips_dummy_area_zero():
           # Dummy area 0 has z-range 0..0, which DOES contain bot_z=0.
           # It must still be excluded.
           g = _mk_graph(_mk_area(1, -10, 10))
           result = areas_on_floor(g, 0.0, z_tolerance=24.0)
           assert 0 not in result
           assert result == [1]

       def test_areas_on_floor_returns_int_indices():
           g = _mk_graph(_mk_area(1, -10, 10), _mk_area(2, -5, 5))
           result = areas_on_floor(g, 0.0, z_tolerance=24.0)
           assert all(isinstance(i, int) for i in result)
           assert sorted(result) == [1, 2]

       def test_areas_on_floor_real_oa_dm1():
           import os, pytest
           path = "maps/oa_dm1.aas"
           if not os.path.exists(path):
               pytest.skip("oa_dm1.aas not present")
           g = parse_aas(path)
           if g is None:
               pytest.skip("failed to parse")
           # Pick a z from the middle of the first real area's z-range
           a1 = g.areas[1]
           mid_z = (a1.mins[2] + a1.maxs[2]) / 2
           subset = areas_on_floor(g, mid_z, z_tolerance=24.0)
           assert len(subset) >= 1
           assert len(subset) < g.num_areas  # multi-story map -> filter must drop some
           assert a1.idx in subset or 1 in subset
       ```
